   136→/* deterministic PRNG so SSR and client render the same sparkline */
   137→function seededPoints(): number[] {
   138→  let s = 7;
   139→  const rand = () => {
   140→    s = (s * 16807) % 2147483647;
   141→    return (s - 1) / 2147483646;
   142→  };
   143→  const pts: number[] = [];
   144→  let v = 146;
   145→  for (let i = 0; i < 44; i++) {
   146→    v += (rand() - 0.47) * 10;
   147→    v = Math.max(128, Math.min(172, v));
   148→    pts.push(v);
   149→  }
   150→  pts[12] = Math.max(pts[12], 166);
   151→  pts[31] = Math.max(pts[31], 169);
   152→  return pts;
   153→}

   155→function buildSpark(pts: number[]) {
   156→  const W = 300;
   157→  const H = 64;
   158→  const P = 5;
   159→  const min = Math.min(...pts);
   160→  const max = Math.max(...pts);
   161→  const span = max - min || 1;
   162→  const step = (W - 2 * P) / (pts.length - 1);
   163→  const xy = pts.map(
   164→    (v, i) => [P + i * step, P + (H - 2 * P) * (1 - (v - min) / span)] as const
   165→  );
   166→  let line = `M ${xy[0][0].toFixed(1)} ${xy[0][1].toFixed(1)}`;
   167→  for (let i = 0; i < xy.length - 1; i++) {
   168→    const p0 = xy[Math.max(0, i - 1)];
   169→    const p1 = xy[i];
   170→    const p2 = xy[i + 1];
   171→    const p3 = xy[Math.min(xy.length - 1, i + 2)];
   172→    const c1x = p1[0] + (p2[0] - p0[0]) / 6;
   173→    const c1y = p1[1] + (p2[1] - p0[1]) / 6;
   174→    const c2x = p2[0] - (p3[0] - p1[0]) / 6;
   175→    const c2y = p2[1] - (p3[1] - p1[1]) / 6;
   176→    line += ` C ${c1x.toFixed(1)} ${c1y.toFixed(1)} ${c2x.toFixed(1)} ${c2y.toFixed(1)} ${p2[0].toFixed(1)} ${p2[1].toFixed(1)}`;
   177→  }
   178→  const area = `${line} L ${xy[xy.length - 1][0].toFixed(1)} ${H} L ${xy[0][0].toFixed(1)} ${H} Z`;
   179→  const ceilingY = P + (H - 2 * P) * (1 - (166 - min) / span);
   180→  return { line, area, last: xy[xy.length - 1], ceilingY };
   181→}

   337→/* chips are evenly spaced on the ring, so they cross the hub intake
   338→   point (left, 180°) in a fixed order — flash each one as it passes */
   339→const PASS_SEQ = [4, 3, 2, 1, 0, 5];
   340→const PASS_MS = 38000 / CLIENTS.length;
   341→
   342→function Orbit({ scale = 1 }: { scale?: number }) {
   343→  const compact = scale < 1;
   344→  const [flash, setFlash] = useState<{
   345→    i: number;
   346→    mode: "served" | "retry";
   347→    count: number;
   348→  } | null>(null);
   349→
   350→  useEffect(() => {
   351→    let idx = 0;
   352→    let count = 0;
   353→    let interval: ReturnType<typeof setInterval> | undefined;
   354→    let retryT: ReturnType<typeof setTimeout> | undefined;
   355→    const fire = () => {
   356→      const i = PASS_SEQ[idx % PASS_SEQ.length];
   357→      idx += 1;
   358→      count += 1;
   359→      if (count % 4 === 0) {
   360→        /* every 4th pass: a 429 gets absorbed, retried, then smooth again */
   361→        setFlash({ i, mode: "retry", count });
   362→        retryT = setTimeout(
   363→          () => setFlash({ i, mode: "served", count: count + 0.5 }),
   364→          550
   365→        );
   366→      } else {
   367→        setFlash({ i, mode: "served", count });
   368→      }
   369→    };
   370→    const first = setTimeout(() => {
   371→      fire();
   372→      interval = setInterval(fire, PASS_MS);
   373→    }, 3400);
   374→    return () => {
   375→      clearTimeout(first);
   376→      if (retryT) clearTimeout(retryT);
   377→      if (interval) clearInterval(interval);
   378→    };
   379→  }, []);
   380→
   381→  return (
   382→    <div
   383→      className="relative"
   384→      style={{ width: 2 * ORBIT_R * scale, height: 2 * ORBIT_R * scale }}
   385→    >
   386→      {/* rings */}
   387→      <div className="absolute inset-0 rounded-full border border-dashed border-stone-900/15" />
   388→      <div className="absolute inset-[17%] rounded-full border border-stone-900/[0.07]" />
   389→      {/* radar dot on inner ring */}
   390→      <div className="core-ring absolute inset-[17%]">
   391→        <span className="absolute left-1/2 top-0 h-1.5 w-1.5 -translate-x-1/2 -translate-y-1/2 rounded-full bg-emerald-400 shadow-[0_0_8px_2px_oklch(0.7_0.14_155/0.45)]" />
   392→      </div>
   393→      {/* hub — the stream plugs in here */}
   394→      <div className="absolute left-1/2 top-1/2 z-10 -translate-x-1/2 -translate-y-1/2">
   395→        {flash ? (
   396→          <span
   397→            key={flash.count}
   398→            aria-hidden
   399→            className="hit-ring absolute left-1/2 top-1/2 h-16 w-16 rounded-2xl border-2 border-emerald-400/60"
   400→          />
   401→        ) : null}
   402→        <div className="hub-pulse flex h-16 w-16 flex-col items-center justify-center rounded-2xl border border-emerald-600/25 bg-white shadow-[0_14px_28px_-12px_oklch(0.6_0.14_155/0.45)]">
   403→          <Terminal
   404→            className={cn("text-emerald-600", compact ? "h-4 w-4" : "h-5 w-5")}
   405→            strokeWidth={1.7}
   406→          />
   407→          <span className="mt-0.5 font-mono text-[7.5px] uppercase tracking-[0.2em] text-stone-400">
   408→            hub
   409→          </span>
   410→        </div>
   411→      </div>
   412→      {/* revolving clients */}
   413→      <div className="orbit-ring absolute inset-0">
   414→        {CLIENTS.map((c, i) => {
   415→          const angle = (i / CLIENTS.length) * Math.PI * 2 - Math.PI / 2;
   416→          const left = `${50 + 50 * Math.cos(angle)}%`;
   417→          const top = `${50 + 50 * Math.sin(angle)}%`;
   418→          return (
   419→            <div
   420→              key={c}
   421→              className="absolute"
   422→              style={{ left, top, transform: "translate(-50%,-50%)" }}
   423→            >
   424→              <div className="orbit-chip">
   425→                <ClientChip
   426→                  name={c}
   427→                  compact={compact}
   428→                  flash={flash?.i === i ? flash.mode : null}
   429→                />
   430→              </div>
   431→            </div>
   432→          );
   433→        })}
   434→      </div>
   435→    </div>
   436→  );
   437→}
