   137→   137→function seededPoints(): number[] {
   138→   138→  let s = 7;
   139→   139→  const rand = () => {
   140→   140→    s = (s * 16807) % 2147483647;
   141→   141→    return (s - 1) / 2147483646;
   142→   142→  };
   143→   143→  const pts: number[] = [];
   144→   144→  let v = 146;
   145→   145→  for (let i = 0; i < 44; i++) {
   146→   146→    v += (rand() - 0.47) * 10;
   147→   147→    v = Math.max(128, Math.min(172, v));
   148→   148→    pts.push(v);
   149→   149→  }
   150→   150→  pts[12] = Math.max(pts[12], 166);
   151→   151→  pts[31] = Math.max(pts[31], 169);
   152→   152→  return pts;

   155→   155→function buildSpark(pts: number[]) {
   156→   156→  const W = 300;
   157→   157→  const H = 64;
   158→   158→  const P = 5;
   159→   159→  const min = Math.min(...pts);
   160→   160→  const max = Math.max(...pts);
   161→   161→  const span = max - min || 1;
   162→   162→  const step = (W - 2 * P) / (pts.length - 1);
   163→   163→  const xy = pts.map(
   164→   164→    (v, i) => [P + i * step, P + (H - 2 * P) * (1 - (v - min) / span)] as const
   165→   165→  );
   166→   166→  let line = `M ${xy[0][0].toFixed(1)} ${xy[0][1].toFixed(1)}`;
   167→   167→  for (let i = 0; i < xy.length - 1; i++) {
   168→   168→    const p0 = xy[Math.max(0, i - 1)];
   169→   169→    const p1 = xy[i];
   170→   170→    const p2 = xy[i + 1];
   171→   171→    const p3 = xy[Math.min(xy.length - 1, i + 2)];
   172→   172→    const c1x = p1[0] + (p2[0] - p0[0]) / 6;
   173→   173→    const c1y = p1[1] + (p2[1] - p0[1]) / 6;
   174→   174→    const c2x = p2[0] - (p3[0] - p1[0]) / 6;
   175→   175→    const c2y = p2[1] - (p3[1] - p1[1]) / 6;
   176→   176→    line += ` C ${c1x.toFixed(1)} ${c1y.toFixed(1)} ${c2x.toFixed(1)} ${c2y.toFixed(1)} ${p2[0].toFixed(1)} ${p2[1].toFixed(1)}`;
   177→   177→  }
   178→   178→  const area = `${line} L ${xy[xy.length - 1][0].toFixed(1)} ${H} L ${xy[0][0].toFixed(1)} ${H} Z`;
   179→   179→  const ceilingY = P + (H - 2 * P) * (1 - (166 - min) / span);
   180→   180→  return { line, area, last: xy[xy.length - 1], ceilingY };
   181→   181→}

   342→   342→function Orbit({ scale = 1 }: { scale?: number }) {
   343→   343→  const compact = scale < 1;
   344→   344→  const [flash, setFlash] = useState<{
   345→   345→    i: number;
   346→   346→    mode: "served" | "retry";
   347→   347→    count: number;
   348→   348→  } | null>(null);

   350→   350→  useEffect(() => {
   351→   351→    let idx = 0;
   352→   352→    let count = 0;
   353→   353→    let interval: ReturnType<typeof setInterval> | undefined;
   354→   354→    let retryT: ReturnType<typeof setTimeout> | undefined;
   355→   355→    const fire = () => {
   356→   356→      const i = PASS_SEQ[idx % PASS_SEQ.length];
   357→   357→      idx += 1;
   358→   358→      count += 1;
   359→   359→      if (count % 4 === 0) {
   360→   360→        /* every 4th pass: a 429 gets absorbed, retried, then smooth again */
   361→   361→        setFlash({ i, mode: "retry", count });
   362→   362→        retryT = setTimeout(
   363→   363→          () => setFlash({ i, mode: "served", count: count + 0.5 }),
   364→   364→          550
   365→   365→        );
   366→   366→      } else {
   367→   367→        setFlash({ i, mode: "served", count });
   368→   368→      }
   369→   369→    };
   370→   370→    const first = setTimeout(() => {
   371→   371→      fire();
   372→   372→      interval = setInterval(fire, PASS_MS);
   373→   373→    }, 3400);
   374→   374→    return () => {
   375→   375→      clearTimeout(first);
   376→   376→      if (retryT) clearTimeout(retryT);
   377→   377→      if (interval) clearInterval(interval);
   378→   378→    };
   379→   379→  }, []);

   395→   395→        {flash ? (
   396→   396→          <span
   397→   397→            key={flash.count}
   398→   398→            aria-hidden
